cmake_minimum_required(VERSION 3.16)
project(addin_client CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(addin_client addin_client.cpp)
target_compile_definitions(addin_client PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(addin_client PRIVATE /utf-8 /W4)
endif()
target_link_libraries(addin_client PRIVATE ole32 oleaut32)
